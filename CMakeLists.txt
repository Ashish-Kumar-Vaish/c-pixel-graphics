cmake_minimum_required(VERSION 3.26)
project(main C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

find_package(SDL3 REQUIRED)

file(GLOB EFFECT_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/effects/*.c")

if(NOT EFFECT_SOURCES)
    message(FATAL_ERROR "No .c files found in effects/ - add at least one.")
endif()

set(REGISTRY_CONTENT "/* Auto-generated by CMake from files in effects/ - do not edit by hand. */\n")
string(APPEND REGISTRY_CONTENT "#include \"effect_registry.h\"\n\n")

set(EFFECT_NAMES "")

foreach(EFFECT_FILE ${EFFECT_SOURCES})
    get_filename_component(EFFECT_NAME ${EFFECT_FILE} NAME_WE)
    string(APPEND REGISTRY_CONTENT "extern void ${EFFECT_NAME}_render(uint32_t frame);\n")
    list(APPEND EFFECT_NAMES ${EFFECT_NAME})
endforeach()

string(APPEND REGISTRY_CONTENT "\nconst Effect g_effects[] = {\n")
foreach(EFFECT_NAME ${EFFECT_NAMES})
    string(APPEND REGISTRY_CONTENT "    { \"${EFFECT_NAME}\", ${EFFECT_NAME}_render },\n")
endforeach()
string(APPEND REGISTRY_CONTENT "};\n\n")

list(LENGTH EFFECT_NAMES EFFECT_COUNT)
string(APPEND REGISTRY_CONTENT "const int g_effect_count = ${EFFECT_COUNT};\n")

file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/effects_registry.c" "${REGISTRY_CONTENT}")

add_executable(main
    main.c
    framebuffer.c
    "${CMAKE_BINARY_DIR}/generated/effects_registry.c"
    ${EFFECT_SOURCES}
)

target_include_directories(main PRIVATE ${CMAKE_SOURCE_DIR})

target_link_libraries(main SDL3::SDL3)
